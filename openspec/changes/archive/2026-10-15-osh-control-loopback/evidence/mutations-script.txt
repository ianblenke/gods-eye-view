def main():
    out, log = [], []
    try:
        for sid, desc, key, old, new in faults:
            path = R0 + F[key]
            text = orig[key].replace(old, new)
            assert text != orig[key], desc
            open(path, 'w').write(text)
            proc = subprocess.Popen(CMD, shell=True, cwd=R0, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, start_new_session=True)
            hung = False
            try:
                text_out, _ = proc.communicate(timeout=120)
            except subprocess.TimeoutExpired:
                hung = True
                os.killpg(proc.pid, signal.SIGKILL)
                text_out, _ = proc.communicate()
            names = set()
            m_exit = re.search(r'EXIT=(\d+)', text_out)
            code = int(m_exit.group(1)) if m_exit else -9
            if hung or code == 124:
                names.add('TIMEOUT (the tests hung)')
            for l in text_out.splitlines():
                l = l.strip()
                if l.startswith('✖ ') and 'failing tests' not in l:
                    n = re.sub(r' \(\d+(\.\d+)?ms\)$', '', l[2:].strip())
                    m = re.match(r'(\[[a-z0-9 -]+\])', n)
                    names.add(m.group(1) if m else 'untagged: ' + n[:60])
            uniq = sorted(names)
            out.append(f'osh-control-{sid} | {desc} | failing: {", ".join(uniq) if uniq else "NONE"}')
            log.append(f'{desc} | exit {code} | failing tests: {", ".join(uniq) if uniq else "NONE"}')
            print(log[-1], flush=True)
            open(path, 'w').write(orig[key])
    finally:
        for key, rel in F.items():
            open(R0 + rel, 'w').write(orig[key])
    for key, rel in F.items():
        assert open(R0 + rel).read() == orig[key], rel
    open(D0 + 'faults.txt', 'w').write('\n'.join(out) + '\n')
    open(D0 + 'faults-run.txt', 'w').write('\n'.join(log) + '\n')

if __name__ == '__main__':
    main()
