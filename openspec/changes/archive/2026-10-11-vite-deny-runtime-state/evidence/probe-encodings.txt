const get = (port, path, extra = {}) => new Promise((resolve) => {
  const req = http.request({ host: '127.0.0.1', port, path, method: 'GET', headers: { Host: `127.0.0.1:${port}`, ...extra }, timeout: 5000 }, (res) => {
    let body = ''; res.on('data', (d) => { if (body.length < 4000) body += d; }); res.on('end', () => resolve({ status: res.statusCode, leak: (body.match(/CANARY-(LOG|OSH|CACHE|PANELKEY|NESTED)/) || [null])[0] }));
  });
  req.on('error', (e) => resolve({ status: 'ERR ' + e.code, leak: null })); req.on('timeout', () => { req.destroy(); resolve({ status: 'TIMEOUT', leak: null }); }); req.end();
});
let patchedLeaks = 0, baseLeaks = 0;
for (const f of forms) {
  const p = await get(5391, f); const b = await get(5392, f);
  if (p.leak) patchedLeaks++; if (b.leak) baseLeaks++;
  console.log(`${String(p.status).padEnd(5)} ${String(p.leak).padEnd(15)} | base ${String(b.status).padEnd(5)} ${String(b.leak).padEnd(15)} | ${f}`);
}
