"""Run each fault of design.md against the two ownership-056 tests; restore each file after."""
import subprocess, shutil
C='scripts/spec/lib/coverage.mjs'; O='scripts/spec/lib/ownership.mjs'
faults = [
 ('filter removed', O, "    changed[file] = changed[file].filter((line) => code.has(line));\n", "    void code;\n"),
 ('script code lines not added', C, "        if (text.trim() !== '') lines.add(first + index);\n", "        void text;\n"),

 ('quoted src taken as src', C, "    if (!SRC_ATTRIBUTE.test(attributes.replace(QUOTED_VALUE, '\"\"'))) {\n      const first", "    if (!SRC_ATTRIBUTE.test(attributes)) {\n      const first"),
 ('line before a line-end attribute', C, "lines.add(lineAt(match.index + 1));", "lines.add(lineAt(match.index));"),
 ('blank script line counted', C, "        if (text.trim() !== '') lines.add(first + index);\n", "        lines.add(first + index);\n"),

    fail = [l for l in r.stdout.splitlines() if l.startswith('ℹ fail')]
