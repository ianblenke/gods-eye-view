F = 'src/locations.js'
orig = open(R + F).read()
START = '  tampa: {\n'
END = '  austin: {\n'
a, b = orig.index(START), orig.index(END)
block = orig[a:b]

def in_block(old, new, nth=1):
    # replace the nth occurrence of old inside the Tampa block only
    idx = -1
    for _ in range(nth):
        idx = block.index(old, idx + 1)
    nb = block[:idx] + new + block[idx + len(old):]
    return orig[:a] + nb + orig[b:]

def move_block(where):
    without = orig[:a] + orig[b:]
    if where == 'after-austin':
        i = without.index('  sf: {\n')
    elif where == 'first':
        i = without.index('  taiwan: {\n')
    return without[:i] + block + without[i:]

cmd = 'node --test --test-force-exit --test-reporter=spec src/locations.test.mjs src/ui/locationControls.test.mjs src/cityPresets.test.mjs'
out, log = [], []
try:
    for sid, desc, fn in faults:
        text = fn()
        assert text != orig, desc
        open(R + F, 'w').write(text)
        p = subprocess.run(cmd, shell=True, cwd=R, capture_output=True, text=True)
        names = set()
        for l in (p.stdout + p.stderr).splitlines():
            l = l.strip()
            if l.startswith('✖ ') and 'failing tests' not in l:
                n = re.sub(r' \(\d+(\.\d+)?ms\)$', '', l[2:].strip())
                m = re.match(r'(\[[a-z-]+-\d+\])', n)
                names.add(m.group(1) if m else 'untagged: ' + n[:60])
        uniq = sorted(names)
        out.append(f'{sid} | {desc} | failing: {", ".join(uniq) if uniq else "NONE"}')
        log.append(f'{desc} | exit {p.returncode} | failing tests: {", ".join(uniq) if uniq else "NONE"}')
        print(log[-1], flush=True)
        open(R + F, 'w').write(orig)
finally:
    open(R + F, 'w').write(orig)
