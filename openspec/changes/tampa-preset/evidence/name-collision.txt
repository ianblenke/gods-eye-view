--- script ---
import { findPoiByName as findBefore, CITY_POIS as POIS_BEFORE } from './src/locationsBeforeTmp.js';
import { findPoiByName as findAfter, CITY_POIS as POIS_AFTER } from './src/locations.js';
import { createPresetGeocoder } from './src/search/presetGeocoder.js';
const gBefore = createPresetGeocoder({ presets: POIS_BEFORE });
const gAfter = createPresetGeocoder({ presets: POIS_AFTER });
const queries = ['Bank of America Plaza', 'Bank of America Plaza in Atlanta', 'fly to Bank of America Plaza, Tampa', 'Bank of America Plaza Tampa', 'Bank of America Plaza, Tampa', 'Raymond James Stadium', 'Ybor City', 'Ybor City Brewing Company'];
const show = (find, pois, r, q) => {
  const m = find(q);
  return `voice ${m ? `${m.cityId}#${m.index} ${pois[m.cityId].pois[m.index].name}` : 'no match'} | geocoder ${r.place ? `${r.place.name} (label: ${r.place.label})` : 'declines'}`;
};
for (const [label, find, pois, g] of [['BEFORE (first place named "Bank of America Plaza", commit 96624f9d)', findBefore, POIS_BEFORE, gBefore], ['AFTER (first place named "Bank of America Plaza, Tampa")', findAfter, POIS_AFTER, gAfter]]) {
  console.log(`## ${label}`);
  for (const q of queries) console.log(`${JSON.stringify(q).padEnd(42)} ${show(find, pois, await g.geocode(q), q)}`);
  console.log('');
}
