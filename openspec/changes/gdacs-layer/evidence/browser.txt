    // Tools-side Vite config for the browser run of gdacs-layer. It is NOT part of the repository.
    // The app gets only a request counter and the GDACS route with a FAKE upstream that answers the
    // stored GDACS page of 2026-10-16 (65 records) and then 204. No request leaves the machine.
    import { readFileSync } from 'node:fs';
    import { defineConfig } from 'vite';
    import { createBrowserViteConfig } from './build/vite.js';
    import { gdacsProxy } from './server/providers/gdacs.js';
    import { apiNotFoundPlugin } from './server/standalone/api-not-found.js';
    
    const PAGE = readFileSync('/home/ianblenke/docker/gev-tools/gdacs-layer/browser/p1.json', 'utf8');
    const counters = { route: 0, upstream: 0, answers: [] };
    
    const fakeFetch = async (url) => {
      counters.upstream += 1;
      const page = new URL(String(url)).searchParams.get('pagenumber');
      if (page === '1')
        return new Response(PAGE, { status: 200, headers: { 'content-type': 'application/json' } });
      return new Response(null, { status: 204 });
    };
    
    const probe = {
      name: 'gd-probe',
      configureServer({ middlewares }) {
        middlewares.use('/__gd', (req, res) => {
          res.setHeader('Content-Type', 'application/json');
          res.end(JSON.stringify(counters));
        });
        middlewares.use((req, res, next) => {
          if ((req.url || '').split('?')[0] === '/api/gdacs') {
            counters.route += 1;
            res.on('finish', () => counters.answers.push(res.statusCode));
          }
          next();
        });
      },
    };
    
    export default defineConfig(({ command }) =>
      createBrowserViteConfig({
        plugins: [
          probe,
          gdacsProxy({ fetchImpl: fakeFetch, now: () => Date.parse('2026-10-16T08:00:00Z') }),
          apiNotFoundPlugin(),
        ],
        host: '127.0.0.1',
        port: 5198,
        command,
      }),
    );
