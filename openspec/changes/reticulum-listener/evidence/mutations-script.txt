out, log = [], []
try:
    for sid, desc, key, old, new, first in faults:
        path = R + FILES[key]
        text = orig[key].replace(old, new, 1) if first else orig[key].replace(old, new)
        assert text != orig[key], desc
        open(path, 'w').write(text)
        proc = subprocess.Popen(cmd, shell=True, cwd=R, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, start_new_session=True)
        hung = False
        try:
            text_out, _ = proc.communicate(timeout=90)
        except subprocess.TimeoutExpired:
            hung = True
            os.killpg(proc.pid, signal.SIGKILL)
            text_out, _ = proc.communicate()
        names = set()
        if hung:
            names.add('TIMEOUT (the tests hung)')
        for l in text_out.splitlines():
            l = l.strip()
            if l.startswith('✖ ') and 'failing tests' not in l:
                n = re.sub(r' \(\d+(\.\d+)?ms\)$', '', l[2:].strip())
                m = re.match(r'(\[[a-z-]+-\d+\])', n)
                names.add(m.group(1) if m else 'untagged: ' + n[:60])
        uniq = sorted(names)
        rc = -9 if hung else proc.returncode
        out.append(f'reticulum-listener-{sid} | {desc} | failing: {", ".join(uniq) if uniq else "NONE"}')
        log.append(f'{desc} | exit {rc} | failing tests: {", ".join(uniq) if uniq else "NONE"}')
        print(log[-1], flush=True)
        open(path, 'w').write(orig[key])
finally:
    for key, rel in FILES.items():
        open(R + rel, 'w').write(orig[key])
for key, rel in FILES.items():
    assert open(R + rel).read() == orig[key], rel
