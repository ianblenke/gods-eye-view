----- leak-preload.mjs -----
// At exit, append the live timer resources of this process to $LEAK_OUT (like the gate's test guard).
import { appendFileSync } from 'node:fs';
process.on('exit', () => {
  const live = (process.getActiveResourcesInfo?.() || []).filter((r) => r === 'Timeout' || r === 'Immediate');
  if (live.length && process.env.LEAK_OUT) appendFileSync(process.env.LEAK_OUT, `${process.argv.slice(1).join(' ').slice(-70)} :: ${live.length} ${[...new Set(live)].join(',')}\n`);
});
