"""Scenario IDs whose test blocks (top-level test(...) up to its first '\n});\n') exist at the base and differ or are gone at HEAD."""
import re, subprocess, sys
base = sys.argv[1]
def git(*a): return subprocess.run(['git', *a], capture_output=True, text=True).stdout
files = [f for f in git('diff', '--name-only', base, 'HEAD').split() if re.search(r'\.test\.(mjs|js)$', f)]
rx = re.compile(r"^\s*test\((['\"`])(.*?)\1", re.M)
IDRX = r'[a-z]+(?:-[a-z]+)*-\d{3}'
def blocks(src):
    out = {}
    for m in rx.finditer(src):
        end = src.find('\n});\n', m.start()); end = len(src) if end < 0 else end + 5
        out[m.group(2)] = src[m.start():end]
    return out
changed, gone = {}, set()
for f in files:
    old, new = git('show', f'{base}:{f}'), git('show', f'HEAD:{f}')
    if not old: continue
    ob, nb = blocks(old), blocks(new)
    for name, body in ob.items():
        if not name.startswith('['): continue
        ids = re.findall(IDRX, name.split(']')[0])
        if name in nb and nb[name] == body: continue
        if name in nb or any(n.startswith('[') and set(re.findall(IDRX, n.split(']')[0])) == set(ids) for n in nb):
            for i in ids: changed.setdefault(i, set()).add(f)
        else:
            gone.update(ids)
print('CHANGED (test edited, scenario kept):', ' '.join(sorted(changed)))
print('GONE (no test with these IDs):', ' '.join(sorted(gone - set(changed))))
for i in sorted(changed): print(' ', i, ' '.join(sorted(changed[i])))
